@loyalid/layout-builder

Drop-in Layout Builder UI + Renderer for CMS. Outputs JSON. Bootstrap 4 & 5.

Install
npm install @loyalid/layout-builder
React
import { useEffect, useRef } from 'react';
import { initLayoutBuilder, generateLayout } from '@loyalid/layout-builder';

export default function LayoutEditor({ value, onChange }) {
  const containerRef = useRef(null);
  const textareaRef = useRef(null);

  useEffect(() => {
    // Create hidden textarea for JSON sync
    const ta = document.createElement('textarea');
    ta.style.display = 'none';
    ta.value = value || '';
    document.body.appendChild(ta);
    textareaRef.current = ta;

    // Init builder
    initLayoutBuilder(containerRef.current, ta);

    // Sync changes back to React state
    const interval = setInterval(() => {
      if (ta.value !== value) onChange(ta.value);
    }, 300);

    return () => {
      clearInterval(interval);
      ta.remove();
    };
  }, []);

  return <div ref={containerRef} style={{ height: 600 }} />;
}

// --- Renderer component ---
import { generateFullHTML } from '@loyalid/layout-builder';

export function LayoutRenderer({ data, bootstrap = 5 }) {
  const html = generateFullHTML(data);
  return <iframe srcDoc={html} style={{ width: '100%', height: 600, border: 'none' }} />;
}
Native JS
<!-- Include the script -->
<script src="/dist/index.js"></script>

<!-- Builder: mount UI, auto-sync JSON to textarea -->
<div id="builder" style="height: 600px;"></div>
<textarea id="layoutData" style="display:none;"></textarea>
<script>
  LoyalidLayout.initLayoutBuilder('#builder', '#layoutData');
</script>

<!-- Renderer: JSON to HTML -->
<script>
  const data = JSON.parse(document.getElementById('layoutData').value);
  const html = LoyalidLayout.generateLayout(data, 5); // Bootstrap 5
  document.getElementById('output').innerHTML = html;
</script>

<!-- Full page -->
<script>
  const page = LoyalidLayout.generateFullHTML(data, { title: 'Article' });
  // page = complete <!DOCTYPE html> string
</script>

API
FunctionDescription
initLayoutBuilder(container, textarea, opts)Mount builder UI, auto-sync JSON to textarea
generateLayout(data, bootstrap)Render JSON to HTML. bootstrap: 0, 4, or 5
generateFullHTML(data, opts)Full standalone HTML page from JSON
Features
  • 3-panel builder UI: Style | Order | Preview
  • Blocks: Title, Paragraph, Image, Divider
  • Inline links: [text](url) in content
  • Convert from HTML: paste editor HTML → auto-convert
  • Drag & drop reorder, move up/down, delete
  • Bootstrap 4 & 5 class support
